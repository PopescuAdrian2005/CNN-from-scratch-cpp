# Development build (keeps assertions enabled)
g++ -std=c++17 -O2 -fopenmp -Wall -Wextra -Wpedantic \
    -Iinclude src/main.cpp -o cnn

# Optimized training build; LTO uses at most 4 jobs
g++ -std=c++17 -O3 -march=native -flto=4 -DNDEBUG \
    -fopenmp -Wall -Wextra -Wpedantic \
    -Iinclude src/main.cpp -o cnn

# CIFAR-10: new model, limited to 4 CPU cores
OMP_NUM_THREADS=4 OMP_DYNAMIC=FALSE \
    taskset -c 0-3 nice -n 19 \
    ./cnn cifar10 weights/cifar10.bin fresh

# CIFAR-10: resume training
OMP_NUM_THREADS=4 OMP_DYNAMIC=FALSE \
    taskset -c 0-3 nice -n 19 \
    ./cnn cifar10 weights/cifar10.bin resume

# MNIST: new model, limited to 4 CPU cores
OMP_NUM_THREADS=4 OMP_DYNAMIC=FALSE \
    taskset -c 0-3 nice -n 19 \
    ./cnn mnist weights/mnist.bin fresh

# MNIST: resume training
OMP_NUM_THREADS=4 OMP_DYNAMIC=FALSE \
    taskset -c 0-3 nice -n 19 \
    ./cnn mnist weights/mnist.bin resume

# Optional: CIFAR-10 with a 12 GB memory limit and no swap
systemd-run --user --scope \
    -p MemoryMax=12G \
    -p MemorySwapMax=0 \
    taskset -c 0-3 \
    nice -n 19 \
    env OMP_NUM_THREADS=4 OMP_DYNAMIC=FALSE \
    ./cnn cifar10 weights/cifar10.bin fresh
